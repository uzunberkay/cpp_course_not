void* operator new(size_t sz)
{
    std::cout << "operator new called sz =  " << sz << '\n';
    if (sz == 1)
        sz = 1;
    void* ptr = std::malloc(sz);
    if (!ptr)
        throw std::bad_alloc{};
    cout << "address of allocated block by :  " << ptr << '\n';

    return ptr;
}
