#include <iostream>
#include <list>
#include <string>

template <typename T>
void cprint(const T& con)
{
    for (const auto& elem : con)
        std::cout << elem << '\n';
}

int main()
{
    using namespace std;

    list<string> x{ "ali", "burak", "can", "emir", "furkan", "ramazan", "yunus" };
    list<string> y{ "banu", "demir", "ramiz", "turgut", "ziya", "zubeyde" };

    x.merge(y);

    cout << "y.size() = " << y.size() << '\n';

    cprint(x);
}
