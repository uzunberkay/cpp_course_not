	template <typename T>
class smartptr
{
public:
	smartptr() = default;
	smartptr(T* p) :mp(p) {}
	~smartptr()
	{
		if (mp)
			delete mp;

	}
	smartptr(smartptr&& other) :mp(other.mp)
	{
		other.mp = nullptr;
	}
	smartptr& operator=(smartptr&& other)
	{
		delete mp;
		mp = other.mp;
		other.mp = nullptr;
		return *this;
	}

	bool has_object()const {
		return mp != nullptr;
	}
	T& operator*()
	{
		return *mp;
	}
	T* operator->()
	{
		return mp;
	}

private:
	T* mp{ nullptr };

};
