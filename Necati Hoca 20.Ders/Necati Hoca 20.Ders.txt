	{
		
		return os <<'[' << m.mval << ']';
	}
	friend std::istream& operator>>(std::istream& is, Mint& m)
	{
		return is >> m.mval;
		
	}
	Mint& operator+=(const Mint& other)
	{
		mval += other.mval;
		return *this;
	}
	friend Mint operator+(const Mint& lhs, const Mint& rhs)
	{
		return Mint(lhs) += rhs;
	}

	Mint& operator-=(const Mint& other)
	{
		mval -= other.mval;
		return *this;
	}

		return Mint(lhs) -= rhs;
