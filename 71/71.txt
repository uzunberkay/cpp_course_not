void worker(std::shared_future<void> start_signal, int id)
{
    std::osyncstream{std::cout} << "Thread " << id << " ready and waiting...\n";

    // Başlangıç sinyalini bekle
    start_signal.wait();

    std::osyncstream{std::cout} << "Thread " << id << " started!\n";

    // Simülasyon amaçlı biraz iş yapalım
    std::this_thread::sleep_for(std::chrono::milliseconds(500));

    std::osyncstream{std::cout} << "Thread " << id << " finished.\n";
}

int main()
{
    std::promise<void> start_promise;

    auto start_signal = start_promise.get_future().share();

    std::vector<std::thread> threads;
    threads.reserve(12);

    for (int i = 1; i <= 10; ++i)
    {
        threads.emplace_back(worker, start_signal, i);
    }

    std::this_thread::sleep_for(std::chrono::seconds(2));

    std::cout << "\nMain thread sending start signal!\n\n";

    start_promise.set_value();

    for (auto& t : threads)
        t.join();
}

int myprint(int id, char c)
{
    using namespace std::chrono_literals;

    std::osyncstream{std::cout} << "thread : " << id << '\n';
    std::this_thread::sleep_for(1000ms);

    int n = rand() * 10;
    int ret = n;

    if (n < 1000)
        n = 2000;

    while (n--)
        std::cout.put(c);

    std::cout.put('\n');

    return ret;
}

{
    using namespace std;

    for (int i = 0; i < 10; ++i) {
        vec[i] = async(launch::async, myprint, i + 'A');
    }

    for (auto& ft : vec)
        cout << ft.get() << '\n';
}

uint64 accummulate_s(const uint64* first, const uint64* last)
{
    return std::accumulate(first, last, 0ull);
}

uint64 acc_parrallel(const std::vector<uint64>& vec)
{
    const auto data = vec.data();
    const auto size = vec.size();

    auto ft1 = std::async(std::launch::async, accummulate_s, data, data + size / 4);
    auto ft2 = std::async(std::launch::async, accummulate_s, data + size / 4, data + 2 * (size / 4));
    auto ft3 = std::async(std::launch::async, accummulate_s, data + 2 * (size / 4), data + 3 * (size / 4));
    auto ft4 = std::async(std::launch::async, accummulate_s, data + 3 * (size / 4), data + size);

    return ft1.get() + ft2.get() + ft3.get() + ft4.get();
}
