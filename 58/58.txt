class Myclass
{
public:
    void* operator new(std::size_t sz)
    {
        std::cout << "Myclass::operator new called sz =  " << sz << '\n';

        return nullptr;
    }
    void operator delete(void*)
    {
        cout << "Myclass::operator delete called!\n";
    }
};

int main()
{
    auto p = new Myclass;


    delete p;
    
}

Myclass::operator new called sz =  1
Myclass::operator delete called!
