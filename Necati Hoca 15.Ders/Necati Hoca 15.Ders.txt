        std::strcpy(mp, p);

    String(const String& other) :mlen(other.mlen), mp(static_cast<char*>(std::malloc(mlen+1)))

        std::cout << "Copy ctor\n";
        std::strcpy(mp, other.mp);

        std::cout << "Copy assignment \n";
