class Myclass{

public:
    Myclass(const Myclass& other); // copy ctor
    Myclass(Myclass&&);  // move ctor

    Myclass& operator=(const Myclass& other); // copy assignment
    Myclass& operator=(Myclass&& other);      // move assignment

}
