class Myclass {
public:
	int foo(int x);
	int bar(int x);
	int baz(int x);
	int bom(int x);
};
using Mfptr = int(Myclass::*)(int);
int main()
{
	Mfptr ar[]={ &Myclass::bar, &Myclass::baz, &Myclass::bom , &Myclass::foo};
	Myclass m;

	for (std::size_t i = 0; i < sizeof(ar) / sizeof(Mfptr); ++i)
	{
		std::cout << (m.*ar[i])(1) << "\n";
	}
	for (auto p : ar)
	{
		std::cout << (m.*p)(1) << "\n";
	}
	

	
}

template <typename T>
auto get_value(T x)
{
	if constexpr (std::is_arithmetic_v<T>)
	{
		return x;
	}
	else if constexpr (std::is_pointer_v<T>)
	{
		return *x;
	}
	else
	{
		return 0;
	}
}
