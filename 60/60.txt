class NameList {
public:
    using container = std::vector<std::string>;

    NameList()
        : sptr_(std::make_shared<container>())
    {
    }

    NameList(std::initializer_list<std::string> list)
        : sptr_(std::make_shared<container>(list))
    {
    }

    void print() const
    {
        for (const auto& s : *sptr_)
            std::cout << s << ' ';

        std::cout << '\n';
    }

    void add(std::string newname)
    {
        sptr_->push_back(std::move(newname));
    }

    void sort()
    {
        std::sort(sptr_->begin(), sptr_->end());
    }

private:
    std::shared_ptr<container> sptr_;
};

int main()
{
    NameList females{ "ayse", "fatma", "nil", "gizem" };

    auto f1 = females;
    auto f2 = females;
    auto f3 = females;
    f1.print();
    f2.print();

    NameList males{ "berkay","kasim","erol" };
    males.print();
}
