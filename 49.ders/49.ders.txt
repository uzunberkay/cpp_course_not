#include <iostream>
#include <vector>
#include <deque>
#include <list>
#include <algorithm>
#include <iterator>
using namespace std;

int main()
{
    vector<int> ivec;

    // ivec içine 0 ile 99 arasında rastgele 40 tane int değer atar.
    rfill(ivec, 40, Irand{ 0, 99 });

    // Orijinal vectorü ekrana yazdırır.
    cprint(ivec);

    // Koşulu sağlayan elemanların kopyalanacağı container.
    deque<int> ideq(100);

    // Koşulu sağlamayan elemanların kopyalanacağı container.
    list<int> ilist(100);

    // partition_copy:
    // ivec içindeki elemanları iki ayrı hedef container'a kopyalar.
    //
    // Predicate true dönerse  -> ideq içine kopyalar.
    // Predicate false dönerse -> ilist içine kopyalar.
    //
    // Buradaki koşul:
    // x % 2 == 0
    //
    // Yani çift sayılar ideq içine,
    // tek sayılar ilist içine kopyalanır.
    auto p = partition_copy(
        ivec.begin(), 
        ivec.end(), 
        ideq.begin(), 
        ilist.begin(),
        [](int x)
        {
            return x % 2 == 0;
        }
    );

    std::cout << "kosulu saglayanlar\n";

    // p.first:
    // ideq içinde kopyalamanın bittiği yeri gösterir.
    // ideq.begin() - p.first arası çift sayılardır.
    copy(ideq.begin(), p.first, ostream_iterator<int>{ cout, " " });
    std::cout << "\n";

    std::cout << "kosulu saglamayanlar\n";

    // p.second:
    // ilist içinde kopyalamanın bittiği yeri gösterir.
    // ilist.begin() - p.second arası tek sayılardır.
    copy(ilist.begin(), p.second, ostream_iterator<int>{ cout, " " });
    std::cout << "\n";
}
