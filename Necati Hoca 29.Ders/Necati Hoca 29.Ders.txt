#include<iostream>
#include<ostream>


class Car {

public:
	void start(){
		std::cout << "Car start\n";
	}
	void stop() {
		std::cout << "Car stop !\n";
	}
	void run()
	{
		std::cout << "Car run!\n";
	}
};

void car_game(Car& car)
{
	car.start();
	car.run();
	car.stop();
}

class Audi:public Car{};
class Bmw:public Car{};
class Volvo:public Car{};
class Honda:public Car{};
class Toyota:public Car{};

int main()
{
	Audi audi;
	Bmw bmw;
	Volvo volvo;

	car_game(audi);
	car_game(bmw);
	car_game(volvo);
	return 0;
}
