class fmtguard {

public:
    fmtguard(std::ostream* p) :m_p(p), m_flags(p->flags()), m_cfill(p->fill()), m_precision(p->precision()),
        m_widht(p->width())
    {
        
    }
    ~fmtguard()
    {
        m_p->flags(m_flags);
        m_p->widht(m_widht);
        m_p->fill(m_cfill);
        m_p->precision(m_precision);
    }


private:
    std::streamsize m_widht;
    char m_cfill;
    int m_precision;
    std::ios::fmtflags m_flags;
    std::ostream* m_p;
};
