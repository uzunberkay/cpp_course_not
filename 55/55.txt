class Counter
{
public:
    Counter() = default;
    Counter(int x) : mc(x) {}

    void increment()
    {
        ++mc;
    }

    void print() const
    {
        std::cout << "[" << mc << "]\n";
    }

private:
    int mc{};
};

int main()
{
    using namespace std;

    vector<Counter> vec{ Counter{10}, Counter{20}, Counter{}, Counter{} };

    for_each(vec.begin(), vec.end(), [](Counter& c) { c.increment(); });
    for_each(vec.begin(), vec.end(), mem_fn(&Counter::increment));
    for_each(vec.begin(), vec.end(), mem_fn(&Counter::print));
}
